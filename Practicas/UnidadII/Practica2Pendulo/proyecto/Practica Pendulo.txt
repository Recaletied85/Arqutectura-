#include <avr/io.h>
.equ LED_PIN, 0

.section .text
.global main

main:
    ; ----- Configurar PB5 como salida -----
    sbi _SFR_IO_ADDR(DDRB), LED_PIN
    sbi _SFR_IO_ADDR(DDRB), 1
    sbi _SFR_IO_ADDR(DDRB), 2

MAIN_LOOP:
    ;==================================
    ; led 1 prender y apagar
    ;==================================
    ; ----- Encender LED -----
    sbi _SFR_IO_ADDR(PORTB), LED_PIN
    rcall DELAY

    ; ----- Apagar LED -----
    cbi _SFR_IO_ADDR(PORTB), LED_PIN
    rcall DELAY
    
    ;==================================
    ; led 2 prender y apagar
    ;==================================
    
    sbi _SFR_IO_ADDR(PORTB), 1
    rcall DELAY

    ; ----- Apagar LED -----
    cbi _SFR_IO_ADDR(PORTB), 1
    rcall DELAY
    
    ;==================================
    ; led 3 prender y apagar 
    ;==================================
    
    sbi _SFR_IO_ADDR(PORTB), 2
    rcall DELAY

    ; ----- Apagar LED -----
    cbi _SFR_IO_ADDR(PORTB), 2
    rcall DELAY
    
     ;=======================================
    ; led 2 se regresa para  prender y apagar 
    ;========================================
    
    sbi _SFR_IO_ADDR(PORTB), 1
    rcall DELAY

    ; ----- Apagar LED -----
    cbi _SFR_IO_ADDR(PORTB), 1
    rcall DELAY

    rjmp MAIN_LOOP

; ============================================================
;  Subrutina de retardo
; ============================================================
DELAY:
    ldi R18, 0x20
DELAY1:
    ldi R19, 0xFF
DELAY2:
    ldi R20, 0xFF
DELAY3:
    subi R20, 1
    brne DELAY3
    subi R19, 1
    brne DELAY2
    subi R18, 1
    brne DELAY1
    ret
